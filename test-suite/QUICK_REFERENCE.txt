🚀 QUICK START (60 seconds)
─────────────────────────────────────────────────────────────────
cd /workspaces/nanolambda/test-suite
./run-all-tests.sh
open http://localhost:8080/dashboard


📋 ALL COMMANDS
─────────────────────────────────────────────────────────────────
./run-all-tests.sh              # Full test suite (~30 min)
./run-tests.sh python sanity    # Python quick test (5 min)
./run-tests.sh nodejs load      # Node.js load test (10 min)
./run-tests.sh java integration # Java integration test
./run-tests.sh mixed integration # All languages together

./stress-test.sh                # High-load test (1 hour)
                                # Tests: 10K req/min, 50 functions

./continuous-monitor.sh start   # Start 24/7 monitoring
./continuous-monitor.sh status  # Check monitoring status
./continuous-monitor.sh stop    # Stop monitoring

./show-results.sh               # View latest results
./show-results.sh 20250101      # View specific run
./show-results.sh --open        # Open dashboard

./cleanup.sh                    # Interactive cleanup
./cleanup.sh --functions        # Delete test functions only
./cleanup.sh --results          # Delete old results
./cleanup.sh --all              # Clean everything
./cleanup.sh --full-reset       # Nuclear option (includes DB)

⚡ EXAMPLE WORKFLOWS
─────────────────────────────────────────────────────────────────
# Daily check
./run-all-tests.sh && ./show-results.sh

# Before production
./run-all-tests.sh
./stress-test.sh
./continuous-monitor.sh start  # Let run for 24h
./show-results.sh

# CI/CD integration
./run-all-tests.sh
if ./show-results.sh | grep -q "100%"; then
    echo "✓ Tests passed"
else
    exit 1
fi

# Development testing
./run-tests.sh python sanity    # Quick Python test
./run-tests.sh nodejs sanity    # Quick Node.js test
./run-tests.sh java sanity      # Quick Java test


🐛 TROUBLESHOOTING
─────────────────────────────────────────────────────────────────

    cargo run --release

Tests failing:
    ./show-results.sh
    cat results/latest/*/output.log

Clean slate:
    ./cleanup.sh --full-reset
    cargo run --release
